## 1. `src/main/resources/db/migration/V7__create_community_posts.sql`

```sql
CREATE TABLE lunchpicker.community_posts (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    author_id BIGINT NOT NULL REFERENCES lunchpicker.users(id),
    body TEXT NOT NULL CHECK (char_length(body) BETWEEN 1 AND 1000),
    area_location public.geography(POINT, 4326) NOT NULL,
    restaurant_id BIGINT REFERENCES lunchpicker.restaurants(id),
    created_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP
);
CREATE INDEX community_posts_area_gix ON lunchpicker.community_posts USING GIST (area_location);
CREATE INDEX community_posts_created_idx ON lunchpicker.community_posts (created_at DESC, id DESC);
```

## 2. `entity/CommunityPost.java`

```java
package com.lunchpicker.community.entity;

import com.lunchpicker.auth.entity.User;
import com.lunchpicker.restaurant.entity.Restaurant;
import jakarta.persistence.*;
import java.time.Instant;
import lombok.AccessLevel;
import lombok.Getter;
import lombok.NoArgsConstructor;
import org.hibernate.annotations.JdbcTypeCode;
import org.hibernate.type.SqlTypes;
import org.locationtech.jts.geom.Point;

@Entity
@Table(name = "community_posts", schema = "lunchpicker")
@Getter
@NoArgsConstructor(access = AccessLevel.PROTECTED)
public class CommunityPost {
    @Id @GeneratedValue(strategy = GenerationType.IDENTITY)
    private Long id;
    @ManyToOne(fetch = FetchType.LAZY, optional = false)
    @JoinColumn(name = "author_id")
    private User author;
    @Column(nullable = false, columnDefinition = "text")
    private String body;
    @JdbcTypeCode(SqlTypes.GEOGRAPHY)
    @Column(nullable = false, columnDefinition = "geography(Point,4326)")
    private Point areaLocation;
    @ManyToOne(fetch = FetchType.LAZY)
    @JoinColumn(name = "restaurant_id")
    private Restaurant restaurant;
    @Column(nullable = false, updatable = false)
    private Instant createdAt;

    // 작성자와 지역, 선택한 식당을 담아 글을 만든다.
    public CommunityPost(User author, String body, Point areaLocation, Restaurant restaurant) {
        this.author = author;
        this.body = body;
        this.areaLocation = areaLocation;
        this.restaurant = restaurant;
        this.createdAt = Instant.now();
    }
}
```

## 5. `repository/CommunityPostRepository.java`

```java
package com.lunchpicker.community.repository;

import com.lunchpicker.community.entity.CommunityPost;
import java.util.List;
import org.springframework.data.jpa.repository.JpaRepository;
import org.springframework.data.jpa.repository.Query;
import org.springframework.data.repository.query.Param;

public interface CommunityPostRepository extends JpaRepository<CommunityPost, Long> {
    // 입력 좌표 3km 안의 최근 글 최대 20개를 조회한다.
    @Query(value = """
            SELECT p.* FROM lunchpicker.community_posts p
            WHERE public.ST_DWithin(p.area_location,
                CAST(public.ST_SetSRID(public.ST_MakePoint(:longitude, :latitude), 4326)
                AS public.geography), 3000)
            ORDER BY p.created_at DESC, p.id DESC
            LIMIT 20
            """, nativeQuery = true)
    List<CommunityPost> findNearby(@Param("latitude") double latitude,
                                   @Param("longitude") double longitude);
}
```

## 6. `service/CommunityPostService.java`

```java
package com.lunchpicker.community.service;

import com.lunchpicker.auth.entity.User;
import com.lunchpicker.auth.repository.UserRepository;
import com.lunchpicker.community.dto.CreatePostRequest;
import com.lunchpicker.community.dto.PostResponse;
import com.lunchpicker.community.entity.CommunityPost;
import com.lunchpicker.community.repository.CommunityPostRepository;
import com.lunchpicker.restaurant.entity.Restaurant;
import com.lunchpicker.restaurant.repository.RestaurantRepository;
import java.util.List;
import lombok.RequiredArgsConstructor;
import org.locationtech.jts.geom.Coordinate;
import org.locationtech.jts.geom.GeometryFactory;
import org.locationtech.jts.geom.PrecisionModel;
import org.locationtech.jts.geom.Point;
import org.springframework.http.HttpStatus;
import org.springframework.http.ProblemDetail;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
import org.springframework.web.ErrorResponseException;

@Service
@RequiredArgsConstructor
public class CommunityPostService {
    private final CommunityPostRepository communityPostRepository;
    private final UserRepository userRepository;
    private final RestaurantRepository restaurantRepository;

    // 로그인한 사용자의 지역 글을 저장한다.
    @Transactional
    public PostResponse create(long authorId, CreatePostRequest request) {
        validateCoordinates(request.getLatitude(), request.getLongitude());
        User author = userRepository.findById(authorId)
                .orElseThrow(() -> error(HttpStatus.UNAUTHORIZED, "로그인이 필요합니다."));
        Restaurant restaurant = null;
        if (request.getRestaurantId() != null) {
            restaurant = restaurantRepository.findById(request.getRestaurantId())
                    .orElseThrow(() -> error(HttpStatus.NOT_FOUND, "음식점을 찾을 수 없습니다."));
        }
        Point point = new GeometryFactory(new PrecisionModel(), 4326)
                .createPoint(new Coordinate(request.getLongitude(), request.getLatitude()));
        CommunityPost post = communityPostRepository.save(new CommunityPost(author, request.getBody().strip(), point, restaurant));
        return toResponse(post);
    }

    // 입력 좌표 3km 안의 최근 글을 조회한다.
    @Transactional(readOnly = true)
    public List<PostResponse> nearby(double latitude, double longitude) {
        validateCoordinates(latitude, longitude);
        return communityPostRepository.findNearby(latitude, longitude).stream().map(this::toResponse).toList();
    }

    // 저장된 글을 API 응답 값으로 변환한다.
    private PostResponse toResponse(CommunityPost post) {
        Restaurant restaurant = post.getRestaurant();
        return new PostResponse(post.getId(), post.getAuthor().getDisplayName(), post.getBody(),
                restaurant == null ? null : restaurant.getId(),
                restaurant == null ? null : restaurant.getName(), post.getCreatedAt());
    }

    // WGS84 위도와 경도 범위를 확인한다.
    private void validateCoordinates(double latitude, double longitude) {
        if (!Double.isFinite(latitude) || latitude < -90 || latitude > 90
                || !Double.isFinite(longitude) || longitude < -180 || longitude > 180) {
            throw error(HttpStatus.BAD_REQUEST, "위도·경도가 유효하지 않습니다.");
        }
    }

    // 상태와 오류 설명을 ProblemDetail로 만든다.
    private ErrorResponseException error(HttpStatus status, String detail) {
        return new ErrorResponseException(status,
                ProblemDetail.forStatusAndDetail(status, detail), null);
    }
}
```

## 7. `controller/CommunityPostController.java`

```java
package com.lunchpicker.community.controller;

import com.lunchpicker.auth.service.AuthService;
import com.lunchpicker.community.dto.CreatePostRequest;
import com.lunchpicker.community.dto.PostResponse;
import com.lunchpicker.community.service.CommunityPostService;
import jakarta.validation.Valid;
import java.util.List;
import lombok.RequiredArgsConstructor;
import org.springframework.http.HttpStatus;
import org.springframework.http.ProblemDetail;
import org.springframework.http.ResponseEntity;
import org.springframework.web.ErrorResponseException;
import org.springframework.web.bind.annotation.*;

@RestController
@RequestMapping("/api/v1/community/posts")
@RequiredArgsConstructor
public class CommunityPostController {
    private final CommunityPostService communityPostService;
    private final AuthService authService;

    // Bearer 토큰으로 작성자를 확인하고 글을 저장한다.
    @PostMapping
    public ResponseEntity<PostResponse> create(
            @RequestHeader(value = "Authorization", required = false) String authorization,
            @Valid @RequestBody CreatePostRequest request) {
        long authorId = authService.me(token(authorization)).getId();
        return ResponseEntity.status(HttpStatus.CREATED).body(communityPostService.create(authorId, request));
    }

    // 선택한 좌표 주변 3km의 글을 반환한다.
    @GetMapping
    public List<PostResponse> nearby(@RequestParam double latitude,
                                     @RequestParam double longitude) {
        return communityPostService.nearby(latitude, longitude);
    }

    // Authorization 헤더에서 Bearer 토큰을 꺼낸다.
    private String token(String authorization) {
        if (authorization == null || !authorization.startsWith("Bearer ")
                || authorization.length() <= 7) {
            throw new ErrorResponseException(HttpStatus.UNAUTHORIZED,
                    ProblemDetail.forStatusAndDetail(HttpStatus.UNAUTHORIZED, "로그인이 필요합니다."), null);
        }
        return authorization.substring(7);
    }
}
```

## 요청 예시

```json
{"body":"오늘은 가락시장역 근처에서 점심 먹어요.","latitude":37.49306,"longitude":127.11806,"restaurantId":null}
```

`POST /api/v1/community/posts`에는 `Authorization: Bearer <로그인 토큰>` 헤더가 필요하다. `GET /api/v1/community/posts?latitude=37.49306&longitude=127.11806`은 로그인 없이 조회할 수 있다. 음식점 자동완성은 기존 `GET /api/v1/restaurants`에 지역 좌표와 `q`를 전달하고, 사용자가 결과를 선택했을 때만 해당 식당 ID를 POST 요청에 넣는다.

현재 구현의 단위 테스트는 `./gradlew test --offline`으로 실행한다. 실제 PostGIS 통합 테스트는 DB 연결 환경에서 `RUN_DB_TESTS=true`로 실행한다.
